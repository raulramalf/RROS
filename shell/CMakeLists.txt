# RROS Shell: el escritorio de RROS (barra, dock, Launchpad, centro de control, notificaciones y fondo) en Qt Quick.
# Funciona sobre KWin (Wayland) en lugar de plasmashell. Las piezas de KDE son opcionales al compilar: sin ellas
# se compila igual y el modo --demo dibuja todo en una ventana (pruebas y capturas).
cmake_minimum_required(VERSION 3.16)
project(rros-shell LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus)
# RROS_SHELL_FULL=ON (la ISO) exige las piezas de KDE: sin ellas el shell no puede ser el escritorio
option(RROS_SHELL_FULL "Exigir LayerShellQt, KWindowSystem y KGlobalAccel" OFF)
if(RROS_SHELL_FULL)
    set(KDE_DEP REQUIRED)
else()
    set(KDE_DEP QUIET)
endif()
find_package(LayerShellQt ${KDE_DEP})
find_package(KF6WindowSystem ${KDE_DEP})
find_package(KF6GlobalAccel ${KDE_DEP})

file(GLOB UI_FILES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} qml/*.qml qml/qmldir)
file(GLOB ICON_FILES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} ../app/icons/*.svg)
add_executable(rros-shell main.cpp shell.cpp shell.h apps.cpp apps.h notifications.cpp notifications.h)
qt_add_resources(rros-shell ui PREFIX "/" FILES ${UI_FILES})
foreach(icon ${ICON_FILES})
    get_filename_component(name ${icon} NAME)
    set_source_files_properties(${icon} PROPERTIES QT_RESOURCE_ALIAS icons/${name})
endforeach()
qt_add_resources(rros-shell icons PREFIX "/" FILES ${ICON_FILES})
target_compile_options(rros-shell PRIVATE -Wall -Wextra)
target_link_libraries(rros-shell PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::DBus)
if(LayerShellQt_FOUND)
    target_link_libraries(rros-shell PRIVATE LayerShellQt::Interface)
    target_compile_definitions(rros-shell PRIVATE HAVE_LAYERSHELL)
endif()
if(KF6WindowSystem_FOUND)
    target_link_libraries(rros-shell PRIVATE KF6::WindowSystem)
    target_compile_definitions(rros-shell PRIVATE HAVE_KWINDOWSYSTEM)
endif()
if(KF6GlobalAccel_FOUND)
    target_link_libraries(rros-shell PRIVATE KF6::GlobalAccel)
    target_compile_definitions(rros-shell PRIVATE HAVE_GLOBALACCEL)
endif()
message(STATUS "RROS Shell: LayerShellQt=${LayerShellQt_FOUND} KWindowSystem=${KF6WindowSystem_FOUND} GlobalAccel=${KF6GlobalAccel_FOUND}")
install(TARGETS rros-shell RUNTIME DESTINATION bin)
